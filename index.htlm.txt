<script>
const LANG_CACHE = JSON.parse(localStorage.getItem('traducciones') || '{}');
let currentLang = localStorage.getItem('lang') || 'es';

// Traduce TODOS los textos visibles de la página
async function setLang(lang, btn) {
  if (lang === currentLang && lang !== 'es') return;
  document.querySelectorAll('.lang-btn').forEach(b => b.classList.remove('active'));
  if (btn) btn.classList.add('active');
  currentLang = lang;
  localStorage.setItem('lang', lang);

  if (lang === 'es') { location.reload(); return; } // vuelve al original

  // Guarda el HTML original la primera vez (para poder volver atrás)
  if (!window._originalHTML) window._originalHTML = document.body.innerHTML;

  // Recolecta todos los nodos de texto traducibles
  const nodes = [];
  const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
    acceptNode(n) {
      if (!n.nodeValue.trim()) return NodeFilter.FILTER_REJECT;
      if (['SCRIPT','STYLE','NOSCRIPT'].includes(n.parentElement.tagName)) return NodeFilter.FILTER_REJECT;
      return NodeFilter.FILTER_ACCEPT;
    }
  });
  while (walker.nextNode()) nodes.push(walker.currentNode);

  const toFetch = [...new Set(nodes.map(n => n.nodeValue.trim()).filter(t => t.length > 1))];

  // Divide en lotes para no saturar la API
  const chunks = [];
  for (let i = 0; i < toFetch.length; i += 40) chunks.push(toFetch.slice(i, i + 40));

  for (const chunk of chunks) {
    const missing = chunk.filter(t => !LANG_CACHE[t + '|' + lang]);
    if (!missing.length) continue;
    try {
      const res = await fetch('https://api.mymemory.translated.net/get', {
        method: 'POST',
        headers: {'Content-Type':'application/x-www-form-urlencoded'},
        body: 'q=' + encodeURIComponent(missing.join(' ||| ')) + '&langpair=es|' + lang
      });
      const data = await res.json();
      const out = (data.responseData.translatedText || '').split(' ||| ');
      missing.forEach((t, i) => { if (out[i]) LANG_CACHE[t + '|' + lang] = out[i]; });
      localStorage.setItem('traducciones', JSON.stringify(LANG_CACHE));
    } catch(e) { console.warn('Fallo traducción lote', e); }
  }

  // Aplica las traducciones
  nodes.forEach(n => {
    const key = n.nodeValue.trim() + '|' + lang;
    if (LANG_CACHE[key]) n.nodeValue = n.nodeValue.replace(n.nodeValue.trim(), LANG_CACHE[key]);
  });
}

// Detecta idioma del navegador al entrar
(function autoDetect(){
  const nav = navigator.language.slice(0,2);
  const soportados = ['en','ro','pt','de','ru','tr'];
  if (soportados.includes(nav) && !localStorage.getItem('lang')) {
    const btn = document.querySelector(`.lang-btn[title][onclick*="'${nav}'"]`);
    if (btn) setTimeout(() => setLang(nav, btn), 800);
  } else if (currentLang !== 'es') {
    const btn = document.querySelector(`.lang-btn[onclick*="'${currentLang}'"]`);
    if (btn) setTimeout(() => setLang(currentLang, btn), 800);
  }
})();

let activeZone='', activePermiso='B';

function selectPermiso(p, btn) {
  activePermiso = p;
  document.querySelectorAll('.permiso-pill').forEach(b => b.classList.remove('active'));
  btn.classList.add('active');
  filterSchools();
}

function getStars(r){const f=Math.floor(r),h=r%1>=0.5?1:0,e=5-f-h;return '★'.repeat(f)+(h?'½':'')+'☆'.repeat(e)}

function renderSchools(list){
  const el=document.getElementById('schoolsList');
  if(!list.length){el.innerHTML='<div class="empty"><div class="empty-icon">🔍</div><p>No se encontraron autoescuelas</p></div>';return}
  el.innerHTML=list.map(s=>{
    const price = s.prices[activePermiso] || s.prices.B;
    const label = activePermiso === 'CAP' ? 'Curso CAP' : `Carnet ${activePermiso}`;
    return `
    <div class="school-card" onclick="openDetail(${s.id})">
      <div class="sc-top"><div class="sc-name">${s.name}</div><div class="sc-price">${price}€ <small>/ ${label}</small></div></div>
      <div class="sc-rating"><span class="sc-stars">${getStars(s.rating)}</span><span class="sc-rating-num">${s.rating}</span><span class="sc-reviews">(${s.reviews})</span></div>
      <div class="sc-zone">📍 ${s.zone}</div>
      <div class="sc-metrics">
        <div class="sc-metric"><div class="val">${s.clase}€</div><div class="lbl">/clase</div></div>
        <div class="sc-metric"><div class="val">${s.aprobados}%</div><div class="lbl">Aprobados</div></div>
        <div class="sc-metric"><div class="val">${s.tasa?'+'+s.tasa+'€':'Gratis'}</div><div class="lbl">Tasa</div></div>
      </div>
      <div class="sc-approve-bar"><div class="sc-approve-fill" style="width:${s.aprobados}%"></div></div>
      <div class="sc-actions">
        <a href="tel:${s.tel}" class="sc-btn sc-btn-accent" onclick="event.stopPropagation()">📱 Llamar</a>
        <button class="sc-btn sc-btn-primary" onclick="event.stopPropagation();openContact(${s.id})">✉️ Contactar</button>
        <button class="sc-btn sc-btn-outline" onclick="event.stopPropagation();openDetail(${s.id})">Ver más</button>
      </div>
    </div>`;
  }).join('');
}

function filterSchools(){
  const q=document.getElementById('searchInput').value.toLowerCase().trim();
  let list=schools.filter(s=>(!activeZone||s.zone===activeZone)&&(!q||s.name.toLowerCase().includes(q)||s.zone.toLowerCase().includes(q)));
  renderSchools(list);
}

function selectZone(z,btn){
  activeZone=z;
  document.querySelectorAll('.zone-pill').forEach(p=>p.classList.remove('active'));
  btn.classList.add('active');
  filterSchools();
}
